import {readFileSync,writeFileSync} from 'node:fs';
import {spawnSync} from 'node:child_process';
import {createHash} from 'node:crypto';
import {join} from 'node:path';
import assert from 'node:assert/strict';
const s=JSON.parse(readFileSync('.tools/android-code23-current.json','utf8'));
const proxy=join('C:/Users/lin0/.codex/parallel-runtimes/qa-hub-preview-7c86','android-recovery-proxy-'+s.runId);

const entries=[...prefs.matchAll(/<string name="([^"]+)">([\s\S]*?)<\/string>/g)].map(m=>({key:decode(m[1]),value:decode(m[2])}));
const pending=entries.filter(e=>e.key.startsWith('pending:')).map(e=>JSON.parse(e.value)).filter(p=>p.scope.projectId===s.projectId);
assert.equal(pending.length,1);const p=pending[0];

assert.equal(p.operationId,before.row.operationId);assert.equal(p.submissionId,state.target.clientSubmissionId);
for(const k of ['accountId','projectId','actorId','installationId','sessionId'])assert.equal(p.scope[k],before.row[k]);
for(const k of ['operationKind','httpMethod','relativePath','payloadJson','idempotencyKey'])assert.equal(p.request[k],before.row[k]);
assert.ok(entries.some(e=>e.key.startsWith('draft:')&&e.value.includes('UNSUBMITTED_CODE22_TO_23_EDIT_65ee5dc8')));

const tappedAt=new Date().toISOString();run(['shell','input','tap','1280','995']);
writeFileSync(join(s.root,'code23-confirm-gesture.json'),JSON.stringify({tappedAt,x:1280,y:995,sourceUi:proof.sourceUi,explicitNativeSubmit:true},null,2)+'\n',{flag:'wx'});
