import {spawnSync} from 'node:child_process';import {readFileSync,writeFileSync,appendFileSync,existsSync} from 'node:fs';import {join} from 'node:path';import assert from 'node:assert/strict';
const s=JSON.parse(readFileSync('.tools/android-code23-current.json','utf8')),root=s.root,adb='C:/Users/lin0/AppData/Local/Android/Sdk/platform-tools/adb.exe',phase=process.argv[2];assert.match(phase,/^[a-z0-9-]+$/);assert.equal(existsSync(join(root,phase+'.xml')),false);assert.equal(existsSync(join(root,phase+'.png')),false);
