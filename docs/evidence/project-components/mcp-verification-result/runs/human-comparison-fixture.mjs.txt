async function fixture(t) {
  const directory = await mkdtemp(join(tmpdir(), "qa-result-automation-result-"));
  const databaseFile = join(directory, "fixture.sqlite");
  const worker = new SqliteStorageWorker({
    databaseFile,
    backupRoot: join(directory, "backups"),
    busyTimeoutMs: 1000,
  });
  const accountId = randomUUID(),
    gmId = randomUUID(),
    projectId = randomUUID(),
    otherProject = randomUUID();
  const bootstrap = {
    accountId,
    actorId: gmId,
    projectId,
    projectKey: "WIRE",
    actorDisplayName: "Fixture GM",
    membershipId: projectMembershipId(projectId, gmId),
    createdAt: new Date().toISOString(),
  };
  await worker.ensureMobileScope(bootstrap);
  await worker.projectManagement({
    accountId,
    actorId: gmId,
    isGm: true,
    operation: "create",
    projectId: otherProject,
    key: "OTHER",
    name: "Other fixture project",
    now: new Date().toISOString(),
  });
  const management = new ProjectManagementService({ accountId, gmUserId: gmId, worker });
  const context = new ProjectRequestContext();
  const scope = context.scope(bootstrap);
  const app = createApiApp({
    logger: false,
    automationPublicApiOrigin: "http://127.0.0.1",
    isolateLegacyComponents: true,
    projectManagementService: management,
    projectRequestContext: context,
    mobileBugStore: createSqliteMobileBugStore({ worker, scope }),
    mobileRelayStore: createSqliteMobileRelayStore({ worker, scope, relayDispatchEnabled: false }),
    mobileVerificationStore: createSqliteMobileVerificationStore({ worker, scope }),
    mobileHumanWorkflowStore: createSqliteMobileHumanWorkflowStore({ worker, scope }),
    browserAuth: {
      store: createSqliteBrowserAuthStore({ worker }),
      accountId,
      userId: gmId,
      actorId: gmId,
      adminEmail: "fixture@example.invalid",
      passwordlessLogin: async () => {
        throw new Error("Unscoped login refused");
      },
      projectLogin: (name, selected, at) => management.login(name, selected, at),
      sessionSecret: "isolated-frozen-wire-fixture",
      cookieName: "qa_frozen_fixture",
      webOrigins: [origin],
      secureCookie: false,
      gm: { userId: gmId, password: "fixture-gm-password" },
    },
  });
  const baseUrl = await app.listen({ host: "127.0.0.1", port: 0 });
  let requestCount = 0;
  t.after(async () => {
    await app.close();
    await worker.close();
    const db = new DatabaseSync(databaseFile, { readOnly: true });
    const schemaVersion = db.prepare("PRAGMA user_version").get().user_version;
    db.close();
    await writeFile(
      join(directory, "retained.json"),
      JSON.stringify({ databaseFile, requestCount, schemaVersion, port: new URL(baseUrl).port }),
    );
    t.diagnostic(
      `retained fixture ${directory}; ${requestCount} real HTTP requests; schema ${schemaVersion}`,
    );
  });
  async function request(
    path,
    {
      body,
      token,
      cookie,
      csrf,
      contentType = vendor,
      project = projectId,
      accept = vendor,
      key,
      method = body ? "POST" : "GET",
      expected = 200,
    } = {},
  ) {
    requestCount++;
    const response = await fetch(`${baseUrl}${path.startsWith("/") ? path : `/api/v1/${path}`}`, {
      method,
      headers: {
        ...(accept ? { accept } : {}),
        origin,
        ...(cookie ? { cookie } : {}),
        ...(csrf ? { "x-csrf-token": csrf } : {}),
        ...(token ? { authorization: `Bearer ${token}` } : {}),
        "x-qa-project-id": project,
        ...(body ? { "content-type": contentType } : {}),
        ...(key ? { "idempotency-key": key } : {}),
      },
      ...(body ? { body: JSON.stringify(body) } : {}),
    });
    const value = await response.json();
    assert.equal(response.status, expected, `${method} ${path}: code=${value.code ?? "response"}`);
    if (path === "auth/login")
      return { ...value, cookie: response.headers.get("set-cookie")?.split(";")[0] };
    return value;
  }
  const employee = await request("auth/login", {
    body: { projectId, name: "History employee", client: "android" },
  });
  const other = await request("auth/login", {
    body: { projectId: otherProject, name: "Other employee", client: "android" },
  });
  const token = employee.accessToken,
    actorId = employee.userId;
  const call = (path, options = {}) => request(path, { token, ...options });
  async function bug(title = "Frozen response regression") {
    const clientSubmissionId = randomUUID();
    const result = await call("bugs", {
      body: {
        submissionContractVersion: "1.1.0",
        clientSubmissionId,
        projectId,
        title,
        description: "Isolated SQLite response fixture",
        expectedBehavior: "Preserve all stored facts",
        severity: "S2",
        priority: "P2",
        attachmentIds: [],
        ownerId: actorId,
        verificationOwnerId: actorId,
        occurrence: {
          observedAt: new Date().toISOString(),
          platform: "web",
          steps: ["Inspect"],
          actualBehavior: "Pending",
        },
      },
      key: `submission:${clientSubmissionId}:commit`,
      expected: 201,
    });
    return result.bug;
  }
  return {
    worker,
    directory,
    employee,
    databaseFile,
    accountId,
    projectId,
    otherProject,
    gmId,
    actorId,
    other,
    call,
    request,
    bug,
  };
}

async function round(f, { criteria = "Check configuration", parent = false } = {}) {
  let bug = await f.bug();
  bug = await f.call(`bugs/${bug.id}/transitions`, {
    body: { expectedVersion: bug.version, toState: "ready" },
    key: `workflow:transitionBug:bug:${bug.id}:v${bug.version}:ready`,
  });
  let attempt = await f.call(`bugs/${bug.id}/repair-attempts`, {
    body: {
      expectedVersion: bug.version,
      mode: "human",
      assigneeId: f.actorId,
      summary: "MCP result human fixture",
    },
    key: `workflow:createRepairAttempt:bug:${bug.id}:v${bug.version}`,
    expected: 201,
  });
  const parentId = parent ? attempt.id : null;
  if (parent) {
    const current = await f.call(`bugs/${bug.id}`);
    await f.call(`bugs/${bug.id}/manual-complete`, {
      body: { expectedVersion: current.version, reason: "Take over the earlier human attempt" },
      key: `workflow:manualCompleteBug:bug:${bug.id}:v${current.version}`,
    });
    attempt = (await f.call(`bugs/${bug.id}/human-workflow`)).repairAttempt;
  } else {
    attempt = await f.call(`repair-attempts/${attempt.id}/start`, {
      body: { expectedVersion: attempt.version },
      key: `workflow:startRepairAttempt:attempt:${attempt.id}:v${attempt.version}`,
    });
    attempt = await f.call(`repair-attempts/${attempt.id}/deliver`, {
      body: {
        expectedVersion: attempt.version,
        deliveryKind: "no_code",
        noCodeReason: "Manual configuration verified",
        summary: "Delivered",
      },
      key: `workflow:deliverRepairAttempt:attempt:${attempt.id}:v${attempt.version}`,
    });
  }
  bug = await f.call(`bugs/${bug.id}`);
  let verification = await f.call(`bugs/${bug.id}/verifications`, {
    body: {
      expectedVersion: bug.version,
      repairAttemptId: attempt.id,
      buildId: null,
      verifierId: f.actorId,
      criteria,
    },
    key: `workflow:createVerification:bug:${bug.id}:attempt:${attempt.id}:v${bug.version}`,
    expected: 201,
  });
  verification = await f.call(`verifications/${verification.id}/start`, {
    body: { expectedVersion: verification.version },
    key: `workflow:startVerification:verification:${verification.id}:v${verification.version}`,
  });
  return {
    bug,
    attempt,
    verification,
    parentId,
    path: `verifications/${verification.id}/result`,
    key: `workflow:recordVerificationResult:verification:${verification.id}:v${verification.version}`,
  };
}
function bodyFor(r, status, legacy = false) {
  return {
    ...(legacy ? {} : { submissionContractVersion: "1.1.0", clientSubmissionId: randomUUID() }),
    expectedVersion: r.verification.version,
    resultSummary: "Original immutable result",
    ...(legacy ? {} : { attachmentIds: [] }),
    status,
    ...(status === "failed" ? { failureReason: "Configuration still fails" } : {}),
    ...(status === "blocked" ? { blockedReason: "Synthetic device temporarily unavailable" } : {}),
  };
}
function snapshot(f) {
  const db = new DatabaseSync(f.databaseFile, { readOnly: true });
  try {
    return Object.fromEntries(
      [
        "bugs",
        "repair_attempts",
        "verifications",
        "events",
        "submissions",
        "idempotency_records",
        "outbox",
        "bug_deletions",
        "verification_result_snapshots",
      ].map((table) => [table, db.prepare(`SELECT * FROM ${table} ORDER BY rowid`).all()]),
    );
  } finally {
    db.close();
  }
}

async function rpc(f, method, params, options = {}) {
  const result = await f.request("/mcp", {
    token: f.employee.accessToken,
    contentType: "application/json",
    accept: "application/json",
    body: { jsonrpc: "2.0", id: randomUUID(), method, params },
    ...options,
  });
  assert.equal(result.jsonrpc, "2.0");
  assert.equal(result.error, undefined);
  return result.result;
}
async function resultTool(f, argumentsValue, { expectedCode, expectedStatus, ...options } = {}) {
  const result = await rpc(
    f,
    "tools/call",
    { name: "qa_record_verification_result", arguments: argumentsValue },
    options,
  );
  assert.equal(result.isError, expectedCode !== undefined);
  const value = result.structuredContent ?? JSON.parse(result.content[0].text);
  assert.deepEqual(JSON.parse(result.content[0].text), value);
  if (expectedCode !== undefined) {
    assert.equal(value.code, expectedCode);
    assert.equal(value.status, expectedStatus);
  }
  return value;
}
const argsFor = (f, r, body) => ({
  projectId: f.projectId,
  verificationId: r.verification.id,
  request: body,
});

for (const status of ["passed", "failed", "blocked"]) {
  test(`actual worker MCP ${status}: canonical first receipt, direct HTTP/shared endpoint parity and immutable replay`, async (t) => {
    const f = await fixture(t),
      r = await round(f, { parent: status === "blocked" }),
      body = bodyFor(r, status),
      args = argsFor(f, r, body);
    const init = await rpc(f, "initialize", {
      protocolVersion: "2025-06-18",
      capabilities: {},
      clientInfo: { name: "real-verification-tool-fixture", version: "1.0" },
    });
    assert.equal(init.protocolVersion, "2025-06-18");
    const before = snapshot(f);
    const first = await resultTool(f, args);
    assert.equal(first.clientSubmissionId, body.clientSubmissionId);
    assert.equal(first.replayed, false);
    assert.equal(first.verification.status, status);
    assert.equal(first.verification.version, r.verification.version + 1);
    assert.equal(
      first.bug.state,
      status === "passed" ? "closed" : status === "failed" ? "ready" : "ready_for_verification",
    );
    assert.deepEqual(
      Object.keys(first).sort(),
      [
        "attachmentIds",
        "bug",
        "captureBundleId",
        "clientSubmissionId",
        "eventId",
        "qaItem",
        "repairAttempt",
        "replayed",
        "verification",
      ].sort(),
    );
    assert.equal(Object.hasOwn(first.verification, "blockedReason"), false);
    const committed = snapshot(f);
    assert.equal(committed.events.length, before.events.length + 1);
    assert.equal(committed.outbox.length, before.outbox.length + 1);
    assert.equal(
      committed.verification_result_snapshots.length,
      before.verification_result_snapshots.length + 1,
    );
    const receipt = committed.idempotency_records.find(
      (x) => x.idempotency_key === r.key && x.operation_id === "recordVerificationResult",
    );
    assert.equal(receipt.actor_id, f.actorId);
    assert.equal(receipt.project_id, f.projectId);
    assert.equal(receipt.audit_event_id, first.eventId);
    assert.deepEqual(await f.call(r.path, { body, key: r.key }), { ...first, replayed: true });
    assert.deepEqual(await resultTool(f, { ...args, idempotencyKey: r.key }), {
      ...first,
      replayed: true,
    });
    assert.deepEqual(
      await f.call("/api/v1/mcp/call", {
        body: { name: "qa_record_verification_result", arguments: args },
      }),
      { ...first, replayed: true },
    );
    assert.deepEqual(snapshot(f), committed);
    const edited = await f.call(`bugs/${r.bug.id}`, {
      method: "PATCH",
      body: {
        expectedVersion: first.bug.version,
        title: "Later edit must not replace original MCP receipt",
      },
      key: `web:updateBug:bug:${r.bug.id}:v${first.bug.version}`,
    });
    const afterEdit = snapshot(f);
    assert.deepEqual(await resultTool(f, args), { ...first, replayed: true });
    assert.deepEqual(snapshot(f), afterEdit);
    await resultTool(
      f,
      { ...args, request: { ...body, resultSummary: "Changed body" } },
      { expectedCode: "IDEMPOTENCY_PAYLOAD_MISMATCH", expectedStatus: 409 },
    );
    assert.deepEqual(snapshot(f), afterEdit);
    if (status === "blocked") {
      assert.deepEqual(committed.repair_attempts, before.repair_attempts);
      assert.equal(committed.bugs[0].active_repair_attempt_id, r.attempt.id);
      assert.equal(committed.bugs[0].active_verification_id, null);
      assert.equal(first.repairAttempt.parentAttemptId, r.parentId);
      assert.equal(
        JSON.parse(committed.verification_result_snapshots[0].response_json).verification
          .blockedReason,
        body.blockedReason,
      );
      let verification = await f.call(`bugs/${r.bug.id}/verifications`, {
        body: {
          expectedVersion: edited.version,
          repairAttemptId: r.attempt.id,
          buildId: null,
          verifierId: f.actorId,
          criteria: "Ready to verify again",
        },
        key: `workflow:createVerification:bug:${r.bug.id}:attempt:${r.attempt.id}:v${edited.version}`,
        expected: 201,
      });
      verification = await f.call(`verifications/${verification.id}/start`, {
        body: { expectedVersion: verification.version },
        key: `workflow:startVerification:verification:${verification.id}:v${verification.version}`,
      });
      const nextBody = bodyFor({ verification }, "passed");
      const closed = await resultTool(f, {
        projectId: f.projectId,
        verificationId: verification.id,
        request: nextBody,
      });
      assert.equal(closed.bug.state, "closed");
      const afterClosed = snapshot(f);
      assert.deepEqual(await resultTool(f, args), { ...first, replayed: true });
      assert.deepEqual(snapshot(f), afterClosed);
      assert.deepEqual(afterClosed.repair_attempts, before.repair_attempts);
    }
  });
}

test("MCP result catalog is additive, explicit and vendor-only; invalid scope/key/body never writes", async (t) => {
  const f = await fixture(t),
    r = await round(f),
    body = bodyFor(r, "blocked"),
    args = argsFor(f, r, body),
    before = snapshot(f);
  const catalog = (await rpc(f, "tools/list", {})).tools;
  assert.equal(catalog.length, 91);
  assert.deepEqual(catalog, AUTOMATION_TOOLS);
  const sharedCatalog = await f.call("/api/v1/mcp/tools");
  assert.deepEqual(sharedCatalog.tools, catalog);
  const definition = catalog.find((x) => x.name === "qa_record_verification_result");
  assert.deepEqual(definition.inputSchema.required, ["projectId", "verificationId", "request"]);
  assert.equal(definition.inputSchema.additionalProperties, false);
  assert.equal(definition.inputSchema.properties.request.additionalProperties, false);
  assert.equal(definition.annotations.idempotentHint, true);
  assert.equal(catalog.filter((x) => x.name !== definition.name).length, 90);
  for (const patch of [
    { idempotencyKey: "not-canonical" },
    { idempotencyKey: r.key.replace(":v2", ":v1") },
    { projectId: "not-a-uuid" },
    { verificationId: "../other/route" },
    { actorId: f.other.userId },
    { bugId: r.bug.id },
    { expectedVersion: 2 },
    { request: { ...body, requireAssignedVerifier: false } },
    { request: { ...body, unknown: "refuse" } },
    { request: { ...body, projectId: f.otherProject } },
    { request: { ...body, clientSubmissionId: undefined } },
    {
      request: {
        expectedVersion: 2,
        status: "passed",
        resultSummary: "Legacy is not this vendor tool",
      },
    },
    { request: { ...body, blockedReason: undefined } },
    { request: { ...body, failureReason: "mixed reason" } },
  ]) {
    await resultTool(
      f,
      { ...args, ...patch },
      { expectedCode: "INVALID_REQUEST", expectedStatus: 400 },
    );
    assert.deepEqual(snapshot(f), before);
  }
  await resultTool(
    f,
    { ...args, request: { ...body, expectedVersion: 1 } },
    { expectedCode: "VERSION_CONFLICT", expectedStatus: 412 },
  );
  assert.deepEqual(snapshot(f), before);
  await resultTool(
    f,
    { ...args, projectId: f.otherProject },
    { expectedCode: "NOT_FOUND", expectedStatus: 404 },
  );
  assert.deepEqual(snapshot(f), before);
  await resultTool(f, args, {
    token: f.other.accessToken,
    expectedCode: "PROJECT_NOT_ACCESSIBLE",
    expectedStatus: 403,
  });
  assert.deepEqual(snapshot(f), before);
});

test("MCP result enforces current assigned verifier, Cookie CSRF, revocation and deletion on first write and replay", async (t) => {
  const f = await fixture(t),
    r = await round(f),
    body = bodyFor(r, "blocked"),
    args = argsFor(f, r, body);
  const peer = await f.request("auth/login", {
    body: { projectId: f.projectId, name: "Different active employee", client: "android" },
  });
  const browser = await f.request("auth/login", {
    body: { projectId: f.projectId, name: "History employee", client: "web" },
  });
  const before = snapshot(f);
  await resultTool(f, args, {
    token: peer.accessToken,
    expectedCode: "FORBIDDEN",
    expectedStatus: 403,
  });
  assert.deepEqual(snapshot(f), before);
  for (const auth of [
    { token: undefined, cookie: browser.cookie },
    { token: undefined, cookie: browser.cookie, csrf: "wrong" },
  ]) {
    // BrowserAuth rejects the outer JSON-RPC HTTP request before dispatch.
    const response = await f.request("/mcp", {
      ...auth,
      contentType: "application/json",
      body: {
        jsonrpc: "2.0",
        id: randomUUID(),
        method: "tools/call",
        params: {
          name: "qa_record_verification_result",
          arguments: args,
        },
      },
      expected: 403,
    });
    assert.equal(response.code, "CSRF_TOKEN_INVALID");
    assert.deepEqual(snapshot(f), before);
  }
  const cookieAuth = { token: undefined, cookie: browser.cookie, csrf: browser.csrfToken };
  const first = await resultTool(f, args, cookieAuth);
  const committed = snapshot(f);
  assert.deepEqual(await resultTool(f, args), { ...first, replayed: true });
  assert.deepEqual(
    await f.call("/api/v1/mcp/call", {
      body: { name: "qa_record_verification_result", arguments: args },
      ...cookieAuth,
    }),
    { ...first, replayed: true },
  );
  await resultTool(f, args, {
    token: peer.accessToken,
    expectedCode: "FORBIDDEN",
    expectedStatus: 403,
  });
  assert.deepEqual(snapshot(f), committed);
  await f.worker.projectManagement({
    accountId: f.accountId,
    actorId: f.gmId,
    isGm: true,
    operation: "membership",
    projectId: f.projectId,
    userId: f.actorId,
    active: false,
    expectedVersion: 1,
    now: new Date().toISOString(),
  });
  const revoked = snapshot(f);
  await resultTool(f, args, {
    ...cookieAuth,
    expectedCode: "PROJECT_NOT_ACCESSIBLE",
    expectedStatus: 403,
  });
  await resultTool(f, args, { expectedCode: "PROJECT_NOT_ACCESSIBLE", expectedStatus: 403 });
  assert.deepEqual(snapshot(f), revoked);

  const fresh = await fixture(t),
    next = await round(fresh),
    request = bodyFor(next, "passed"),
    nextArgs = argsFor(fresh, next, request);
  const accepted = await resultTool(fresh, nextArgs);
  await fresh.call(`bugs/${next.bug.id}?expectedVersion=${accepted.bug.version}`, {
    method: "DELETE",
    key: `web:deleteBug:bug:${next.bug.id}:v${accepted.bug.version}`,
  });
  const deleted = snapshot(fresh);
  await resultTool(fresh, nextArgs, { expectedCode: "NOT_FOUND", expectedStatus: 404 });
  assert.deepEqual(snapshot(fresh), deleted);
});

test("ordinary MCP human closure retains employee permissions and does not reinterpret blocked as an old action", async (t) => {
  const f = await fixture(t),
    r = await round(f),
    body = bodyFor(r, "passed");
  const peer = await f.request("auth/login", {
    body: { projectId: f.projectId, name: "Ordinary human closure peer", client: "android" },
  });
  const before = snapshot(f);
  await resultTool(f, argsFor(f, r, body), {
    token: peer.accessToken,
    expectedCode: "FORBIDDEN",
    expectedStatus: 403,
  });
  const humanArguments = {
    projectId: f.projectId,
    bugId: r.bug.id,
    verificationId: r.verification.id,
    expectedVersion: r.verification.version,
    idempotencyKey: r.key,
    request: body,
  };
  const rejected = await rpc(
    f,
    "tools/call",
    {
      name: "qa_bug_action",
      arguments: { ...humanArguments, action: "blocked" },
    },
    { token: peer.accessToken },
  );
  assert.equal(rejected.isError, true);
  assert.equal(JSON.parse(rejected.content[0].text).code, "INVALID_REQUEST");
  assert.equal(JSON.parse(rejected.content[0].text).status, 400);
  assert.deepEqual(snapshot(f), before);
  const accepted = await rpc(
    f,
    "tools/call",
    {
      name: "qa_bug_action",
      arguments: { ...humanArguments, action: "verify_pass" },
    },
    { token: peer.accessToken },
  );
  assert.equal(accepted.isError, false);
  const value = accepted.structuredContent;
  assert.deepEqual(JSON.parse(accepted.content[0].text), value);
  assert.equal(value.bug.state, "closed");
  assert.equal(value.result.verification.verifierId, f.actorId);
  assert.equal(value.result.bug.id, r.bug.id);
  const committed = snapshot(f);
  assert.equal(committed.events.length, before.events.length + 1);
  assert.equal(committed.events.at(-1).actor_id, peer.userId);
  assert.equal(committed.verification_result_snapshots.at(-1).actor_id, peer.userId);
  assert.equal(committed.verifications.at(-1).verifier_id, f.actorId);
  await resultTool(f, argsFor(f, r, body), {
    token: peer.accessToken,
    expectedCode: "FORBIDDEN",
    expectedStatus: 403,
  });
  assert.deepEqual(snapshot(f), committed);
});
