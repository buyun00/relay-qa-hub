import fs from 'node:fs';
import assert from 'node:assert/strict';
import {createHash} from 'node:crypto';
import {spawnSync} from 'node:child_process';
const base='docs/evidence/project-components';
const dir=base+'/coverage-parallel-map-20260909';
const actual=base+'/coverage-regeneration-live/6e77e27e-cc84-4fcb-92f5-2175eb0dd9be';
const sha=b=>createHash('sha256').update(b).digest('hex');
const write=(p,s)=>fs.writeFileSync(p,s,{flag:'wx'});
const mapper='scripts/project-components/map-coverage-evidence.mjs';
const audit='scripts/project-components/parallel-evidence-map.audit.mjs';
assert.equal(sha(fs.readFileSync(mapper)),'2ff1c4bbfcd8df06e1c093c32cbfde6f8c5f3182fbcc40dfb3a3328db5e35938');
const checks=[];
for(const [name,args] of [
 ['audit-final',[audit,'--expected-mapper',sha(fs.readFileSync(mapper))]],
 ['syntax-mapper-final',['--check',mapper]],['syntax-audit-final',['--check',audit]],
 ['lint-final',['node_modules/eslint/bin/eslint.js',mapper,audit]],
 ['format-final',['node_modules/prettier/bin/prettier.cjs','--check',mapper,audit]],
]){
 const startedAt=new Date().toISOString();
 const r=spawnSync(process.execPath,args,{encoding:'utf8',windowsHide:true,maxBuffer:10*1024*1024});
 write(dir+'/'+name+(name==='audit-final'?'.json':'.txt'),r.stdout??'');
 write(dir+'/'+name+'.stderr.txt',r.stderr??'');
 const check={name,args,startedAt,finishedAt:new Date().toISOString(),exitCode:r.status};
 checks.push(check);write(dir+'/'+name+'-exit.json',JSON.stringify(check,null,2)+'\n');
 assert.equal(r.status,0,name);
}
write(dir+'/final-checks.json',JSON.stringify(checks,null,2)+'\n');
write(dir+'/audit-current-source.mjs.txt',fs.readFileSync(audit));
write(dir+'/final-check-runner.mjs.txt',fs.readFileSync('.tools/finish-parallel-coverage-final.mjs'));
write(actual+'/refresh-coverage-protected.mjs.txt',fs.readFileSync('.tools/refresh-coverage-protected.mjs'));
console.log(JSON.stringify({checks:checks.length,allPassed:true,audit:sha(fs.readFileSync(audit))}));
